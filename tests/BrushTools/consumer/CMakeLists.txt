cmake_minimum_required(VERSION 3.31)
project(iiBrushToolsConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 ON)
find_package(iiPaintEngine 0.5 CONFIG REQUIRED COMPONENTS BrushTools)
find_package(iiBrushTools 0.1 CONFIG REQUIRED)
add_executable(consumer ../ContractTest.cpp)
target_link_libraries(consumer PRIVATE iiPaintEngine::BrushTools iiBrushTools::iiBrushTools)
enable_testing()
foreach(_case Controls Taper Correction Parameters Input Tiles)
    add_test(NAME Installed.${_case} COMMAND consumer ${_case})
    set_tests_properties(Installed.${_case} PROPERTIES ENVIRONMENT_MODIFICATION
        "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;LD_LIBRARY_PATH=unset:")
endforeach()

foreach(_contract PresetContract PresetAvailability PreviewQueueContract PreviewQueueAllocation FrozenSource)
    add_executable(consumer_${_contract} ../${_contract}Test.cpp)
    target_link_libraries(consumer_${_contract} PRIVATE iiBrushTools::iiBrushTools)
    if(_contract STREQUAL "FrozenSource")
        find_package(Threads REQUIRED)
        target_link_libraries(consumer_${_contract} PRIVATE Threads::Threads)
    endif()
    add_test(NAME Installed.${_contract} COMMAND consumer_${_contract})
    set_tests_properties(Installed.${_contract} PROPERTIES ENVIRONMENT_MODIFICATION
        "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;LD_LIBRARY_PATH=unset:")
endforeach()
